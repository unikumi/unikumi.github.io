// A fő lista Gist nyers (raw) URL-je const LIST_URL = "https://gist.githubusercontent.com/unikumi/2b35bd479eb40c8bc6ff06e57f8658b0/raw/gistfile1.txt"; // Egyszerű, natív adatletöltő (CORS proxy nélkül!) async function fetchJson(url) { const res = await fetch(url); if (!res.ok) throw new Error(`Hiba a letöltés során: ${res.status}`); return await res.json(); } // Térkép inicializálása const map = new maplibregl.Map({ container: 'map', style: 'https://tiles.openfreemap.org/styles/liberty', center: [16.0, 49.0], zoom: 8 }); map.addControl(new maplibregl.NavigationControl({ showCompass: true }), 'bottom-right'); let layerList = []; let currentMarkers = []; let userLocation = null; let userHeading = 0; let isFollowing = false; const userMarkerEl = document.createElement('div'); userMarkerEl.className = 'user-marker'; let userMarker = null; function fixGeojson(data) { if (!data || typeof data !== 'object') return { type: "FeatureCollection", features: [] }; if (data.type === "FeatureCollection") { (data.features || []).forEach(f => { if (!f.properties) f.properties = {}; }); } return data; } function fitDataBounds(geojson, markers) { const bounds = new maplibregl.LngLatBounds(); let count = 0; if (geojson && geojson.features) { geojson.features.forEach(f => { if (f.geometry && f.geometry.type === "LineString") { f.geometry.coordinates.forEach(coord => { bounds.extend(coord); count++; }); } }); } markers.forEach(m => { if (m.lat !== undefined && m.lng !== undefined) { bounds.extend([m.lng, m.lat]); count++; } }); if (count > 0 && !isFollowing) { map.fitBounds(bounds, { padding: 40 }); } } // Réteg betöltése Gist URL-ekről async function loadLayer(index) { const item = layerList[index]; if (!item) return; try { const [rawGeojson, markersData] = await Promise.all([ fetchJson(item.geojson), fetchJson(item.markers) ]); const geojsonData = fixGeojson(rawGeojson); if (map.getSource('route-source')) { map.getSource('route-source').setData(geojsonData); } else { map.addSource('route-source', { type: 'geojson', data: geojsonData }); map.addLayer({ id: 'route-layer', type: 'line', source: 'route-source', paint: { 'line-color': '#0b3d91', 'line-width': 2 } }); } currentMarkers.forEach(m => m.remove()); currentMarkers = []; (markersData || []).forEach(pt => { if (pt.lat === undefined || pt.lng === undefined) return; const el = document.createElement('div'); el.className = 'marker-container'; el.innerHTML = `
${pt.name || ''}
`; const marker = new maplibregl.Marker({ element: el, anchor: 'left' }) .setLngLat([pt.lng, pt.lat]) .addTo(map); currentMarkers.push(marker); }); fitDataBounds(geojsonData, markersData || []); } catch (err) { console.error("Hiba a réteg betöltésekor:", err); } } function updatePositionAndBearing() { if (!userLocation) return; if (!userMarker) { userMarker = new maplibregl.Marker({ element: userMarkerEl }) .setLngLat(userLocation) .addTo(map); } else { userMarker.setLngLat(userLocation); } userMarkerEl.style.transform = `rotate(${userHeading}deg)`; if (isFollowing) { map.jumpTo({ center: userLocation, bearing: userHeading }); } } // Geolocation if ('geolocation' in navigator) { navigator.geolocation.watchPosition( (pos) => { userLocation = [pos.coords.longitude, pos.coords.latitude]; if (pos.coords.heading !== null && !isNaN(pos.coords.heading)) { userHeading = pos.coords.heading; } updatePositionAndBearing(); }, (err) => console.warn("GPS hiba:", err), { enableHighAccuracy: true } ); } // Gyroskóp / Kompassz function handleOrientation(e) { let heading = null; if (e.webkitCompassHeading !== undefined) { heading = e.webkitCompassHeading; } else if (e.alpha !== null) { heading = 360 - e.alpha; } if (heading !== null) { userHeading = heading; updatePositionAndBearing(); } } const btnFollow = document.getElementById('btnFollow'); btnFollow.addEventListener('click', async () => { isFollowing = !isFollowing; btnFollow.classList.toggle('active', isFollowing); if (isFollowing) { if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') { try { const resp = await DeviceOrientationEvent.requestPermission(); if (resp === 'granted') { window.addEventListener('deviceorientation', handleOrientation, true); } } catch (e) { console.error(e); } } else { window.addEventListener('deviceorientationabsolute', handleOrientation, true) || window.addEventListener('deviceorientation', handleOrientation, true); } updatePositionAndBearing(); } else { map.setBearing(0); } }); // Indítás map.on('load', async () => { try { layerList = await fetchJson(LIST_URL); const dropdown = document.getElementById('layerSelect'); } });
